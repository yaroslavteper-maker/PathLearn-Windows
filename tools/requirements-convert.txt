# Dependencies for tools/convert_extractor.py only.
#
# PathLearn itself never imports torch, timm or transformers — it runs ONNX.
# Keep this in a SEPARATE venv from the app:
#
#     python -m venv ~/.venvs/pathlearn-convert
#     ~/.venvs/pathlearn-convert/Scripts/pip install -r requirements-convert.txt
#
# CPU torch is deliberate. Conversion is a handful of forward passes on random
# input; the GPU build is a multi-gigabyte download that buys nothing here, and
# the app's inference GPU support is a separate concern (onnxruntime-gpu).
# If you already have a CUDA torch, it works fine too — nothing below pins it.
#
# Versions are lower bounds taken from a venv that successfully converted
# phikon-v1, uni-v1 and uni2-h. Newer is expected to work; the export is
# verified against PyTorch (cosine >= 0.9999) before a descriptor is written,
# so a version that breaks the graph fails loudly rather than silently.

torch>=2.4
timm>=1.0.9
transformers>=4.44
safetensors>=0.4
huggingface_hub>=0.24
onnx>=1.16
onnxruntime>=1.18
numpy>=1.26
